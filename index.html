  <script>

    const CLAIMS_URL = "https://444669a7-f30d-433d-b040-6e30f8af2637.usrfiles.com/ugd/444669_0136e423c6d940eeaae0d6b74fcb47ab.json";

    const PROOFS_URL = "https://444669a7-f30d-433d-b040-6e30f8af2637.usrfiles.com/ugd/444669_1808839a4f844d90b9de50804129c0c3.json";

    const CONTRACT_ADDRESS = "0xf8E567D269B9C9be82520A6c8Cf76f364aa2735d";

    const ABI = [{

      "inputs":[

        {"internalType":"uint256","name":"qty","type":"uint256"},

        {"internalType":"uint256","name":"maxFor","type":"uint256"},

        {"internalType":"bytes32[]","name":"proof","type":"bytes32[]"}

      ],

      "name":"claim",

      "outputs":[],

      "stateMutability":"nonpayable",

      "type":"function"

    }];


    let claims = {}, proofs = {}, signer, address;


    const connectBtn = document.getElementById('connect');

    const claimBtn = document.getElementById('claim');

    const msg = document.getElementById('msg');

    const addrDiv = document.getElementById('address');

    const claimable = document.getElementById('claimable');

    const qtyInput = document.getElementById('qty');


    function show(m, c="black"){ msg.style.color=c; msg.textContent=m; }

    async function loadData(){

      claims = await (await fetch(CLAIMS_URL)).json();

      proofs = (await (await fetch(PROOFS_URL)).json()).proofs || {};

    }

    async function connect(){

      if(!window.ethereum) return show("No wallet detected","red");

      const provider = new ethers.BrowserProvider(window.ethereum);

      signer = await provider.getSigner();

      address = (await signer.getAddress()).toLowerCase();

      addrDiv.textContent = address;

      const allowed = claims[address] || 0;

      claimable.value = allowed;

      show(allowed>0?"Wallet is eligible ✅":"Not eligible ❌", allowed>0?"green":"red");

    }

    async function claim(){

      if(!signer) return connect();

      const allowed = claims[address] || 0;

      const qty = Number(qtyInput.value);

      if(qty < 1 || qty > allowed) return show("Invalid quantity","red");

      const proof = proofs[address] || [];

      const c = new ethers.Contract(CONTRACT_ADDRESS, ABI, signer);

      show("Sending transaction...","blue");

      try {

        const tx = await c.claim(qty, allowed, proof);

        await tx.wait();

        show("✅ Claimed successfully!","green");

      } catch(e){ show("❌ "+(e.reason||e.message),"red"); }

    }


    connectBtn.onclick = connect;

    claimBtn.onclick = claim;

    loadData();

  </script>
